\documentclass[border=5pt]{standalone}
\usepackage{circuitikz}
\usetikzlibrary{calc}
\begin{document}
\begin{circuitikz}
  \node[op amp] (A) {};
  % cadena de entrada al + (abajo)
  \draw (A.+) -- ++(-0.4,0) coordinate(b);
  \draw (b) to[R=$R_2$] ++(-1.9,0) coordinate(a);
  \draw (a) to[R=$R_1$] ++(-1.9,0) to[short,-o] ++(-0.3,0) node[left]{$v_i$};
  % C2 de b a tierra
  \draw (b) to[C=$C_2$] ++(0,-1.7) node[ground]{};
  % realimentacion - a la salida (arriba)
  \draw (A.-) -- ++(0,0.9) coordinate(t) -| (A.out);
  % C1 del nodo a a la salida (ruta superior)
  \draw (a) -- ++(0,2.3) coordinate(ct);
  \draw (ct) to[C=$C_1$] (ct -| A.out) coordinate(cr);
  \draw (cr) -- (A.out);
  % salida
  \draw (A.out) to[short,-o] ++(1.3,0) node[right]{$v_o$};
