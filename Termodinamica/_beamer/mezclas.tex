%  RESUMEN
% ============================================================
\begin{frame}{Resumen}
\begin{cardidea}{Síntesis}{\faLightbulb}
Una \textbf{mezcla} de gases combina componentes por sus \textbf{fracciones}; sus propiedades aparentes ($M$, $R$, $c_p$) salen de promediar por $y_i$. El \textbf{aire húmedo} es la mezcla aire seco + vapor: se describe con seis magnitudes ($\omega,\phi,T_d,T_{bh},h,v$) ligadas por fórmulas sencillas y leídas de un vistazo en la \textbf{carta psicrométrica}. La \textbf{combustión} transforma reactivos en productos: se balancea, se mide el exceso de aire y se calcula el calor liberado y la temperatura de llama.
\end{cardidea}
\vspace{2mm}
\begin{columns}[T]
\begin{column}{0.485\textwidth}
\begin{cardgear}{Psicrometría}{\faTint}
Todo estado del aire se fija con \textbf{dos} propiedades; las cinco restantes se leen en la carta a $P=101.325$ kPa. Base de los balances: por kg de \textbf{aire seco}.
\end{cardgear}
\end{column}
\begin{column}{0.485\textwidth}
\begin{cardgear}{Combustión}{\faFire}
$\mathrm{C_xH_y}+\text{aire}\to$ productos. El \textbf{exceso de aire} baja la temperatura de llama; el poder calorífico mide el calor liberado por kg.
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

\portadaseccion{1}{Mezclas de gases ideales}{Dalton · Amagat · fracciones · propiedades}{\faLayerGroup}

% ---- Dalton y Amagat ----
\begin{frame}{Dalton, Amagat y fracciones}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\centering
\includegraphics[width=\linewidth,height=3.9cm,keepaspectratio]{mezcla_gases_dalton_esquema}\\[0.5mm]
\fsub{Cada gas ocupa todo $V$ a la misma $T$ y aporta su presión parcial.}
\end{column}
\begin{column}{0.5\textwidth}
\begin{cardidea}{Composición}{\faSuperscript}
Fracción molar $y_i=\dfrac{n_i}{n}$;\ \ másica $fm_i=\dfrac{m_i}{m}$.\\[0.8mm]
$\displaystyle\sum_i y_i=\sum_i fm_i=1,\qquad fm_i=\dfrac{y_i M_i}{M}$.
\end{cardidea}
\vspace{1.1mm}
\begin{cardgear}{Dalton y Amagat}{\faInfoCircle}
\textbf{Dalton} (presiones): $P_i=y_i P,\ \ P=\sum_i P_i$.\\[0.6mm]
\textbf{Amagat} (volúmenes): $V_i=y_i V,\ \ \tfrac{V_i}{V}=y_i$.\\[0.6mm]
El análisis volumétrico da directo las $y_i$.
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- Propiedades de la mezcla ----
\begin{frame}{Propiedades aparentes de la mezcla}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{cardidea}{Masa molar y $R$}{\faSuperscript}
$M=\dfrac{m}{n}=\displaystyle\sum_i y_i M_i=\dfrac{1}{\sum_k fm_k/M_k}$\\[0.9mm]
$R=\dfrac{R_u}{M}$\ \ (constante de la mezcla).
\end{cardidea}
\vspace{1.1mm}
\begin{cardgear}{Energía, entalpía, $c_p$}{\faInfoCircle}
Sólo función de $T$ (gas ideal):\\[0.5mm]
$\bar{u}=\sum_i y_i\bar{u}_i,\quad \bar{h}=\sum_i y_i\bar{h}_i$\\[0.6mm]
$\bar{c}_p=\sum_i y_i\bar{c}_{p,i},\quad \gamma=\bar{c}_p/\bar{c}_v$.\\[0.6mm]
Entropía: a presiones parciales $P_i$.
\end{cardgear}
\end{column}
\begin{column}{0.46\textwidth}
\begin{cardnota}[\faCalculator]
\textbf{Ejemplo (gas de síntesis).} $30\%$CO, $25\%$H$_2$, $15\%$CO$_2$, $30\%$N$_2$.\\[0.6mm]
$M=0.30(28.01)+0.25(2.016)+0.15(44.01)+0.30(28.01)$\\[0.4mm]
$\Rightarrow\ M=23.91\ \mathrm{kg/kmol}$.\\[0.6mm]
$\bar{c}_p=31.18\ \mathrm{kJ/(kmol\,K)}$; comprimir a $500$ K, $800$ kPa da $\Delta h=260.8\ \mathrm{kJ/kg}$.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

\portadaseccion{2}{Fórmulas de aire húmedo}{$\omega$ · $\phi$ · $T_d$ · $T_{bh}$ · $h$}{\faTint}

% ---- Introducción aire húmedo ----
\begin{frame}{El aire húmedo: seis magnitudes}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.4cm,keepaspectratio]{diagrama_psicrometrico}\\[0.5mm]
\fsub{El estado del aire húmedo vive en el plano $(T,\omega)$.}
\end{column}
\begin{column}{0.46\textwidth}
\begin{cardidea}{Idea base}{\faInfoCircle}
El aire húmedo es aire seco $+$ vapor de agua. Se describe con $T$, $\omega$, $\phi$, $T_d$, $T_{bh}$, $h$ (y $v$). Todo balance se plantea \textbf{por kg de aire seco}, porque $\dot m_a$ es lo que se conserva.
\end{cardidea}
\vspace{1.3mm}
\begin{cardnota}[\faInfoCircle]
El vapor se trata como gas ideal: su presión parcial $P_v$ es la variable que conecta casi todas las fórmulas. La presión total de referencia es $P=101.325$ kPa.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- omega ----
\begin{frame}{Humedad específica (razón de humedad) $\omega$}
\begin{cardidea}{Definición y fórmula}{\faSuperscript}
Masa de vapor por masa de aire seco. Con $M_a=28.97$, $M_v=18.015$ ($\Rightarrow 0.622=M_v/M_a$):
$$\omega \equiv \frac{m_v}{m_a} = \frac{n_v M_v}{n_a M_a} = 0.622\,\frac{P_v}{P - P_v}\quad\left[\tfrac{\text{kg vapor}}{\text{kg a.s.}}\right],\qquad P_v=\frac{\omega\,P}{0.622+\omega}.$$
\end{cardidea}
\vspace{1.6mm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{cardgear}{Qué relaciona}{\faInfoCircle}
Mide \textbf{cuánto vapor} lleva el aire. Es la ordenada de la carta. Sube al humidificar, baja al condensar. Sólo depende de $P_v$ y $P$, no de $T$.
\end{cardgear}
\end{column}
\begin{column}{0.46\textwidth}
\begin{cardnota}[\faCalculator]
$T=20\,^\circ$C, $\phi=50\%$: $P_v=1.169$ kPa\\[0.5mm]
$\omega=0.622\times\dfrac{1.169}{101.325-1.169}=0.00726$.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- phi ----
\begin{frame}{Humedad relativa $\phi$ y su relación con $\omega$}
\begin{cardidea}{Definición y fórmula}{\faSuperscript}
Cociente entre el vapor presente y el máximo posible a esa $T$ (saturación):
$$\phi \equiv \frac{P_v}{P_{\rm sat}(T)}\in[0,1],\qquad\Longrightarrow\qquad \omega = 0.622\,\frac{\phi\,P_{\rm sat}(T)}{P - \phi\,P_{\rm sat}(T)}.$$
\end{cardidea}
\vspace{1.6mm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{cardgear}{Lectura física}{\faInfoCircle}
$\phi=100\%$ es aire saturado; $\phi=0$ aire seco. A $\omega$ fija, \textbf{calentar baja $\phi$} (crece $P_{\rm sat}$ pero $P_v$ no): por eso las curvas $\phi=$cte divergen a la derecha.
\end{cardgear}
\end{column}
\begin{column}{0.46\textwidth}
\begin{cardnota}[\faExclamationTriangle]
$\phi$ y $\omega$ \textbf{no} son lo mismo: dos aires con igual $\omega$ tienen $\phi$ distinta si difiere $T$. $\phi$ mide confort; $\omega$ mide masa de agua.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- T rocío ----
\begin{frame}{Temperatura de rocío $T_d$}
\begin{cardidea}{Definición y fórmula}{\faSuperscript}
Temperatura a la que el aire, enfriado a $\omega$ y $P$ constantes, alcanza la saturación y \textbf{empieza a condensar}:
$$P_{\rm sat}(T_d) = P_v = \frac{\omega\,P}{0.622+\omega}\qquad\Longrightarrow\qquad T_d = T_{\rm sat}\big(P_v\big).$$
\end{cardidea}
\vspace{1.6mm}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{cardgear}{Dónde condensa}{\faInfoCircle}
Es la temperatura de las superficies frías donde aparece rocío o empañamiento. En la carta se lee trazando una \textbf{horizontal} ($\omega$ cte) desde el estado hasta $\phi=100\%$.
\end{cardgear}
\end{column}
\begin{column}{0.44\textwidth}
\begin{cardnota}[\faCalculator]
$\omega=0.00726$: $P_v=1.169$ kPa\\[0.5mm]
$\Rightarrow T_d=T_{\rm sat}(1.169)\approx 9.3\,^\circ$C.\\[0.5mm]
Cuanto más cerca esté $T_d$ de $T$, mayor $\phi$.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- T bulbo húmedo ----
\begin{frame}{Temperatura de bulbo húmedo $T_{bh}$ y saturación adiabática}
\begin{cardidea}{Definición y fórmula}{\faSuperscript}
$T_{bh}$ es la temperatura del \textbf{saturador adiabático}: el aire se humidifica hasta saturar tomando calor del propio flujo. Con $c_{pa}=1.005$ y $\omega_{bh}$ el $\omega$ de saturación a $T_{bh}$:
$$\omega = \omega_{bh} - \frac{c_{pa}\,(T - T_{bh})}{h_{fg}(T_{bh})},\qquad \omega_{bh}=0.622\,\frac{P_{\rm sat}(T_{bh})}{P-P_{\rm sat}(T_{bh})}.$$
\end{cardidea}
\vspace{1.5mm}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{cardgear}{Cómo se mide}{\faInfoCircle}
Se lee con un termómetro de \textbf{bulbo mojado} ventilado (psicrómetro). Siempre $T_d\le T_{bh}\le T$; la diferencia $T-T_{bh}$ crece cuando el aire está más seco.
\end{cardgear}
\end{column}
\begin{column}{0.44\textwidth}
\begin{cardnota}[\faCalculator]
$T=35$, $T_{bh}=25\,^\circ$C: $\omega_{bh}=0.02011$;\\[0.5mm]
$\omega=0.02011-\dfrac{1.005(35-25)}{2442}=0.0160$.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- entalpia ----
\begin{frame}{Entalpía del aire húmedo $h$ y grado de saturación}
\begin{cardidea}{Definición y fórmula}{\faSuperscript}
Energía por kg de aire seco: sensible del aire $+$ del vapor. Con $T$ en $^\circ$C, $c_{pv}=1.86$, $h_{fg,0}=2501$:
$$h = c_{pa}\,T + \omega\,(h_{fg,0}+c_{pv}T) = (1.005 + 1.86\,\omega)\,T + 2501\,\omega\quad[\tfrac{\text{kJ}}{\text{kg a.s.}}].$$
\end{cardidea}
\vspace{1.5mm}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{cardgear}{Uso y grado de saturación}{\faInfoCircle}
$h$ es la magnitud de los balances de energía HVAC. El \textbf{grado de saturación} $\mu=\omega/\omega_{\rm sat}(T)$ es afín a $\phi$ pero medido en base a $\omega$.
\end{cardgear}
\end{column}
\begin{column}{0.44\textwidth}
\begin{cardnota}[\faCalculator]
$T=20\,^\circ$C, $\omega=0.00726$:\\[0.5mm]
$h=(1.005+1.86\times0.00726)20+2501\times0.00726$\\[0.4mm]
$=20.37+18.15=38.5$ kJ/kg a.s.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Mezcla adiabática de dos corrientes}
\begin{columns}[T]

El estado de mezcla divide $\overline{1\text{-}2}$ en razón \textbf{inversa} a los caudales: $\dfrac{d(1\text{-}3)}{d(2\text{-}3)}=\dfrac{\dot m_{a2}}{\dot m_{a1}}$. Si el segmento cruza la saturación, aparece niebla.

% ---- Concepto ----
\begin{frame}{Torres: contacto directo aire--agua}
\begin{columns}[T]
\begin{column}{0.42\textwidth}
\centering
\includegraphics[width=\linewidth,height=5.0cm,keepaspectratio]{torre_enfriamiento_esquema}\\[0.3mm]

\begin{column}{0.54\textwidth}
\begin{cardidea}{Qué es}{\faInfoCircle}
Dispositivo de \textbf{contacto directo}: el agua caliente se enfría cediendo vapor al aire. No es un intercambiador cerrado; ocurre transferencia de \textbf{calor y de masa} a la vez.
\end{cardidea}
\vspace{1.0mm}
\begin{cardgear}{Por qué enfría tanto}{\faTint}
El grueso del enfriamiento es \textbf{evaporativo}: cada kg de agua evaporada se lleva $\approx2400$ kJ de calor latente. Límite físico: el agua no baja de $T_{bh,1}$ (bulbo húmedo del aire que entra).
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- Balances torres ----
\begin{frame}{Torres: balances de masa y energía}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{cardidea}{Variables}{\faListUl}
\footnotesize
$\dot m_a$: aire seco (se conserva). $\dot m_3,\dot m_4$: agua caliente/fría. $\omega_1,\omega_2$: humedad del aire a la entrada/salida. $h_{a1},h_{a2}$: entalpías del aire. $h_f(T)$: entalpía del agua líquida.
\end{cardidea}
\vspace{0.9mm}
\begin{cardgear}{Balances (VC estacionario, $\dot Q=0$)}{\faSuperscript}
\footnotesize
Aire seco: $\dot m_a=$ cte.\\[0.4mm]
Agua: $\dot m_4=\dot m_3-\dot m_a(\omega_2-\omega_1)$.\\[0.5mm]
Energía: $\dfrac{\dot m_3}{\dot m_a}=\dfrac{(h_{a2}-h_{a1})-(\omega_2-\omega_1)h_f(T_4)}{h_f(T_3)-h_f(T_4)}$.
\end{cardgear}
\end{column}
\begin{column}{0.44\textwidth}
\begin{cardidea}{Agua de reposición}{\faPlusCircle}
El agua evaporada $=$ ganancia de humedad del aire, y se repone (maquillaje):
$$\dot m_{\rm ev}=\dot m_{\rm maq}=\dot m_a(\omega_2-\omega_1).$$
\end{cardidea}
\vspace{0.9mm}
\begin{cardnota}[\faRulerHorizontal]
\textbf{Rango} $=T_3-T_4$: cuánto baja el agua.\\[0.4mm]
\textbf{Aproximación} $=T_4-T_{bh,1}$: cercanía al límite ($\lesssim5\,^\circ$C es excelente).
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

\begin{columns}[T]
\begin{column}{0.44\textwidth}

\begin{column}{0.52\textwidth}
\begin{cardidea}{Reacción estequiométrica}{\faSuperscript}
\footnotesize
$\mathrm{C_xH_y}+a_{\rm est}(\mathrm{O_2}+3.76\,\mathrm{N_2})\longrightarrow$\\[0.4mm]
$\qquad x\,\mathrm{CO_2}+\tfrac{y}{2}\mathrm{H_2O}+3.76\,a_{\rm est}\,\mathrm{N_2}$
\end{cardidea}
\vspace{0.8mm}
\begin{cardgear}{Balanceo (se conservan los átomos)}{\faListOl}
\footnotesize
\textbf{C}: $x\to x$, da $x$ de $\mathrm{CO_2}$.\\[0.3mm]
\textbf{H}: $y\to y$, da $y/2$ de $\mathrm{H_2O}$.\\[0.3mm]
\textbf{O}: $2a_{\rm est}=2x+\tfrac{y}{2}\Rightarrow a_{\rm est}=x+\tfrac{y}{4}$.\\[0.3mm]
\textbf{N}: entra $=$ sale ($\mathrm{N_2}$ inerte).\\[0.5mm]
Ej.: $\mathrm{CH_4}\Rightarrow a_{\rm est}=2$; $\mathrm{C_3H_8}\Rightarrow a_{\rm est}=5$.
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- AF y equivalencia ----
\begin{frame}{Relación aire--combustible y equivalencia}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{cardidea}{Relación aire-combustible $AF$}{\faBalanceScale}
Aire suministrado por unidad de combustible. Dos bases:
$$\underbrace{AF=\frac{m_{\rm aire}}{m_{\rm comb}}}_{\text{másica }[\text{kg/kg}]},\qquad \underbrace{\overline{AF}=\frac{n_{\rm aire}}{n_{\rm comb}}}_{\text{molar}}.$$
Conversión: $AF=\overline{AF}\,\dfrac{M_{\rm aire}}{M_{\rm comb}}$ (con $M_{\rm aire}\!\approx\!28.97$).\\[0.7mm]
\textbf{Estequiométrico $AF_{\rm est}$}: el $AF$ con el \emph{aire teórico exacto} (combustión completa, sin exceso ni defecto). Para $C_xH_y$: $a_{\rm est}=x+\tfrac{y}{4}$ mol $\mathrm{O_2}$, aire $=a_{\rm est}(1{+}3.76)$ mol.
\end{cardidea}
\end{column}
\begin{column}{0.48\textwidth}
\begin{cardgear}{Exceso de aire y $\Phi$}{\faSlidersH}
\textbf{Relación de equivalencia} $\Phi=\dfrac{AF_{\rm est}}{AF}$:\\[0.5mm]
$\Phi<1$ \textbf{pobre}: exceso de $\mathrm{O_2}$.\\[0.3mm]
$\Phi=1$ estequiométrica.\\[0.3mm]
$\Phi>1$ \textbf{rica}: aparece CO.\\[0.5mm]
Exceso de aire $e=\left(\dfrac{1}{\Phi}-1\right)\times100\%$.
\end{cardgear}
\vspace{0.8mm}
\begin{cardnota}[\faExclamationTriangle]
$AF$ es una relación de masas; $\Phi$ es adimensional. Más aire $\Rightarrow$ mayor $AF\Rightarrow$ menor $\Phi$ (más pobre).
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- Balance de energía ----
\begin{frame}{Balance de energía: formación y poder calorífico}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{cardidea}{Entalpía de formación $\bar h_f^\circ$}{\faSuperscript}
\footnotesize
Entalpía para formar 1 kmol desde sus elementos a $25\,^\circ$C, 1 atm. Elementos estables: $\bar h_f^\circ=0$.\\[0.5mm]
Calor de reacción:\\
$\bar h_R^\circ=\sum_{\rm prod}n_i\bar h_{f,i}^\circ-\sum_{\rm react}n_j\bar h_{f,j}^\circ<0$.
\end{cardidea}
\vspace{0.8mm}
\begin{cardnota}[\faInfoCircle]
\footnotesize

\end{cardnota}
\end{column}
\begin{column}{0.48\textwidth}
\begin{cardgear}{Poder calorífico (por kg comb.)}{\faFire}
\footnotesize
$\mathrm{PCI}=-\bar h_R^\circ$ con $\mathrm{H_2O}$ \textbf{vapor}.\\[0.4mm]
$\mathrm{PCS}=-\bar h_R^\circ$ con $\mathrm{H_2O}$ \textbf{líquida}.\\[0.4mm]
$\mathrm{PCS}-\mathrm{PCI}=\dfrac{n_{\rm H_2O}M_{\rm H_2O}}{M_{\rm comb}}\,h_{fg}$.
\end{cardgear}
\vspace{0.8mm}
\begin{cardnota}[\faCalculator]
\footnotesize

\end{cardnota}
\end{column}
\end{columns}
