\begin{frame}{Resumen}
\begin{cardidea}{Síntesis}{\faLightbulb}
Toda la termodinámica de procesos se apoya en \textbf{cuatro balances} sobre una región de análisis: \textbf{masa} (se conserva), \textbf{energía} (se conserva, 1.\textsuperscript{a} ley), \textbf{entropía} (se \emph{genera}, 2.\textsuperscript{a} ley: $S_{gen}\ge0$) y \textbf{exergía} (se \emph{destruye}: $X_{dest}=T_0S_{gen}\ge0$). Los escribimos para un \textbf{sistema cerrado} (masa fija) y para un \textbf{volumen de control} (flujo). El vínculo maestro es \textbf{Gouy--Stodola}: cada irreversibilidad genera entropía y destruye trabajo útil.
\end{cardidea}
\vspace{2mm}
\begin{columns}[T]
\begin{column}{0.485\textwidth}
\begin{cardgear}{Lo que se conserva}{\faCogs}
\textbf{Masa} y \textbf{energía} nunca se crean ni destruyen: sus balances se \emph{cierran} sumando entradas, salidas y acumulación.
\end{cardgear}
\end{column}
\begin{column}{0.485\textwidth}
\begin{cardgear}{Lo que se degrada}{\faRecycle}
La \textbf{entropía} solo puede generarse y la \textbf{exergía} solo puede destruirse: miden la \emph{dirección} y la \emph{calidad} del proceso.
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ============================================================
%  SECCIÓN 1 — SISTEMA CERRADO
% ============================================================
\portadaseccion{1}{Sistema Cerrado (SC)}{masa fija · $Q$ · $W$ · $\Delta U$ · $S_{gen}$ · exergía}{\faThermometerHalf}

% ---- Primera ley SC ----
\begin{frame}{Primera ley --- balance de energía (SC)}
\begin{columns}[T]
\begin{column}{0.44\textwidth}
\centering
\includegraphics[width=\linewidth,height=2.9cm,keepaspectratio]{sistema_cerrado_frontera}\\[1mm]
\includegraphics[width=\linewidth,height=3.0cm,keepaspectratio]{primera_ley_SC_diagrama_Pv}\\[0.4mm]
\fsub{Frontera cerrada y trabajo $=$ área bajo $P$-$v$.}
\end{column}
\begin{column}{0.52\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Para masa fija, $Q-W$ es función de estado (Joule):
$$\Delta U = Q - W,\qquad dU=\delta Q-\delta W.$$
$Q>0$ entra; $W>0$ sale. Trabajo de frontera: $W_b=\int_1^2 P\,dV$.
\end{cardidea}
\vspace{1.2mm}
\begin{cardgear}{Casos particulares}{\faSuperscript}
{\footnotesize
Isocórico: $\Delta U=Q_v$.\\[0.4mm]
Isobárico: $Q_P=\Delta H$.\\[0.4mm]
Adiabático: $\Delta U=-W$.\\[0.4mm]
Ciclo: $Q_{neto}=W_{neto}$.\\[0.4mm]
Gas ideal: $\Delta u=c_v\,\Delta T$.}
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- Segunda ley / entropía SC ----
\begin{frame}{Segunda ley --- balance de entropía (SC)}
\begin{cardidea}{Balance de entropía}{\faRandom}
Se \emph{deduce} de la desigualdad de Clausius $\oint \delta Q/T_b\le0$. Introduce un déficit no negativo, la entropía generada $S_{gen}$:
$$S_2-S_1=\int_1^2\frac{\delta Q}{T_b}+S_{gen},\qquad S_{gen}\ge0.$$
La 1.\textsuperscript{a} ley dice \emph{cuánta} energía cambia; la 2.\textsuperscript{a} ley impone la \textbf{dirección}: $S_{gen}=0$ solo en el límite reversible.
\end{cardidea}
\vspace{1.4mm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{cardgear}{Cómo calcular $S_{gen}$}{\faCalculator}
{\footnotesize
$S$ es función de estado: se integra $\delta Q_{rev}/T$ por un camino reversible auxiliar.
$$S_{gen}=\Delta S-\int_1^2\frac{\delta Q_{real}}{T_b}\ge0.$$
Adiabático: $\Delta S=S_{gen}\ge0$ (isentrópico si reversible).}
\end{cardgear}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardnota}[\faInfoCircle]
\textbf{Principio de incremento de entropía.} La del sistema puede bajar (si sale calor), pero la del \emph{universo} (sistema $+$ entorno) nunca decrece. Fuentes de $S_{gen}$: fricción, mezcla, calor con $\Delta T$ finito, reacción.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- Exergía SC ----
\begin{frame}{Balance de exergía (SC)}
\begin{cardidea}{Exergía y su balance}{\faBalanceScale}
La \textbf{exergía} $B$ es el trabajo útil máximo hasta el estado muerto $(T_0,P_0)$. Para un sistema cerrado:
$$B=(U-U_0)+P_0(V-V_0)-T_0(S-S_0).$$
$$\Delta B=\int_1^2\!\Big(1-\tfrac{T_0}{T}\Big)\delta Q-\big[W-P_0\Delta V\big]-X_{dest}.$$
\end{cardidea}
\vspace{1.2mm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{cardgear}{Trabajo útil}{\faSuperscript}
{\footnotesize
Al expandirse, $P_0\Delta V$ solo empuja la atmósfera:
$$W_{\acute util}=W-P_0\Delta V,\quad W_{\acute util,\max}=-\Delta B.$$
Aislado ($Q{=}0,W{=}0$): $\Delta B=-X_{dest}\le0$.}
\end{cardgear}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardnota}[\faInfoCircle]
\textbf{Gouy--Stodola:} $X_{dest}=T_0\,S_{gen}\ge0$. La energía se conserva, pero la exergía \emph{siempre} disminuye o se conserva. Reducir pérdidas $=$ reducir irreversibilidades.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ============================================================
\portadaseccion{2}{Volumen de Control (VC)}{flujo · $\dot m$ · $\dot Q$ · $\dot W$ · $\dot S_{gen}$ · $\psi$}{\faCogs}

% ---- Balance de masa VC ----
\begin{frame}{Balance de masa (VC)}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\centering
\includegraphics[width=\linewidth,height=3.3cm,keepaspectratio]{balance_masa_VC}\\[0.6mm]
\fsub{Entradas y salidas de un VC; el desbalance acumula masa.}
\end{column}
\begin{column}{0.5\textwidth}
\begin{cardidea}{Conservación de la masa}{\faInfoCircle}
$$\frac{dm_{VC}}{dt}=\sum_i\dot m_i-\sum_e\dot m_e.$$
Flujo másico: $\dot m=\rho V A=\dfrac{VA}{v}$ \,[kg/s].
\end{cardidea}
\vspace{1.2mm}
\begin{cardgear}{Régimen estacionario}{\faSuperscript}
{\footnotesize
Continuidad: $\sum_i\dot m_i=\sum_e\dot m_e$, es decir $\rho_1V_1A_1=\rho_2V_2A_2$.\\[0.5mm]
Incompresible: $V_1A_1=V_2A_2$.\\[0.5mm]
Transitorio: $m_2-m_1=\sum m_i-\sum m_e$.}
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- Balance de energía VC ----
\begin{frame}{Balance de energía --- SFSS (VC)}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\centering
\includegraphics[width=\linewidth,height=3.3cm,keepaspectratio]{balance_energia_VC_SFSS}\\[0.6mm]
\fsub{VC estacionario: entra $h_1$, sale $h_2$; eje $\dot W$.}
\end{column}
\begin{column}{0.5\textwidth}
\begin{cardidea}{Primera ley del VC}{\faInfoCircle}
El flujo transporta \textbf{entalpía} $h=u+Pv$ (incluye el trabajo de flujo):
$$\dot Q-\dot W=\sum_e\dot m_e\Big(h+\tfrac{V^2}{2}+gz\Big)_e-\sum_i\dot m_i\Big(h+\tfrac{V^2}{2}+gz\Big)_i.$$
\end{cardidea}
\vspace{1.0mm}
\begin{cardgear}{Dispositivos (1 e / 1 s)}{\faSuperscript}
{\footnotesize
Turbina: $w=h_1-h_2>0$.\\[0.4mm]
Compresor/bomba: $w=h_2-h_1$ (entra).\\[0.4mm]
Tobera/difusor: $h_1+\tfrac{V_1^2}{2}=h_2+\tfrac{V_2^2}{2}$.\\[0.4mm]
Válvula: $h_1=h_2$.}
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- Balance de entropía VC ----
\begin{frame}{Balance de entropía (VC)}
\begin{cardidea}{Segunda ley del VC}{\faRandom}
La entropía entra/sale con el calor ($\dot Q_k/T_k$) y con el flujo ($\dot m\,s$), y siempre se \emph{genera} dentro:
$$\frac{dS_{VC}}{dt}=\sum_k\frac{\dot Q_k}{T_k}+\sum_i\dot m_i s_i-\sum_e\dot m_e s_e+\dot S_{gen},\qquad \dot S_{gen}\ge0.$$
\end{cardidea}
\vspace{1.4mm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{cardgear}{Estacionario (1 e / 1 s)}{\faSuperscript}
{\footnotesize
$$\dot S_{gen}=\dot m(s_e-s_i)-\sum_k\frac{\dot Q_k}{T_k}\ge0.$$
Adiabático: $s_e\ge s_i$ (nunca decrece).\\[0.4mm]
Ideal: $s_e=s_i$ (isentrópico).}
\end{cardgear}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardnota}[\faInfoCircle]
\textbf{Eficiencia isentrópica} (referencia $s_2=s_1$):\\[0.6mm]
Turbina $\eta_T=\dfrac{h_1-h_2}{h_1-h_{2s}}$;\;
compresor $\eta_C=\dfrac{h_{2s}-h_1}{h_2-h_1}$.\\[0.6mm]
Vínculo con exergía: $\dot X_{dest}=T_0\dot S_{gen}$.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- Balance de exergía VC ----
\begin{frame}{Balance de exergía (VC)}
\begin{cardidea}{Exergía de flujo y su balance}{\faBalanceScale}
La corriente ya pagó su trabajo de flujo $\Rightarrow$ transporta la \textbf{exergía de flujo} $\psi$ (con $h$, no $u$):
$$\psi=(h-h_0)-T_0(s-s_0)+\tfrac{V^2}{2}+gz.$$
$$\dot W_{\acute util}=\dot m(\psi_1-\psi_2)+\sum_k\!\Big(1-\tfrac{T_0}{T_k}\Big)\dot Q_k-\dot X_{dest}.$$
\end{cardidea}
\vspace{1.2mm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{cardgear}{Destrucción de exergía}{\faSuperscript}
{\footnotesize
Gouy--Stodola: $\dot X_{dest}=T_0\,\dot S_{gen}\ge0$.\\[0.5mm]
Gas ideal:\\ $\psi=c_p(T-T_0)-T_0\!\big[c_p\ln\tfrac{T}{T_0}-R\ln\tfrac{P}{P_0}\big]$.}
\end{cardgear}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardnota}[\faInfoCircle]
\textbf{Eficiencia exergética.} Turbina $\epsilon_T=\dfrac{\dot W}{\dot m(\psi_1-\psi_2)}$; compresor $\epsilon_C=\dfrac{\dot m(\psi_2-\psi_1)}{\dot W}$. Localiza \emph{dónde} se pierde potencial de trabajo.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

%  SECCIÓN 3 — ENTROPÍA Y EXERGÍA EN PROFUNDIDAD
% ============================================================
\portadaseccion{3}{Entropía y Exergía}{Clausius · estado muerto · Gouy--Stodola · $\eta_{II}$}{\faRandom}

% ---- ¿Qué es la entropía? ----
\begin{frame}{¿Qué es la entropía?}
\begin{cardidea}{Definición}{\faInfoCircle}
La \textbf{entropía} $S$ mide la dispersión del estado microscópico: cuántas formas hay de repartir la energía entre los grados de libertad sin cambiar el estado macroscópico. Es la propiedad que hace \emph{exacta} la forma $\delta Q_{rev}/T$:
$$dS=\frac{\delta Q_{rev}}{T}\quad[\text{kJ/K}].$$
\end{cardidea}
\vspace{1.2mm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{cardgear}{Desigualdad de Clausius}{\faSuperscript}
{\footnotesize
Para todo ciclo, con $=$ si es reversible:
$$\oint\frac{\delta Q}{T_b}\le0.$$
De aquí, $\displaystyle\int_1^2\!\frac{\delta Q_{rev}}{T}$ no depende del camino $\Rightarrow$ define $S$ como función de estado.}
\end{cardgear}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardnota}[\faInfoCircle]
\textbf{Calidad del calor.} No es lo mismo 100~J a 1000~K que a 300~K: la razón $\delta Q/T$ distingue calor de alta y baja calidad. Ecuaciones $T\,ds$: $T\,ds=du+P\,dv=dh-v\,dP$.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- Entropía estadística y generación ----
\begin{frame}{Entropía estadística y generación}
\begin{columns}[T]
\begin{column}{0.45\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.0cm,keepaspectratio]{entropia_microestados_expansion}\\[0.5mm]
\fsub{Expansión libre: $\Omega_2\gg\Omega_1\Rightarrow\Delta S>0$.}
\end{column}
\begin{column}{0.51\textwidth}
\begin{cardidea}{Boltzmann: $S=k_B\ln\Omega$}{\faRandom}
Más microestados accesibles $\Rightarrow$ mayor entropía. La ``flecha del tiempo'' es la tendencia estadística de baja $\Omega$ a alta $\Omega$.
\end{cardidea}
\vspace{1.0mm}
\begin{cardgear}{Significado de $S_{gen}$}{\faInfoCircle}
{\footnotesize
$S_{gen}>0$ mide la \textbf{irreversibilidad} interna: fricción, mezcla, calor con $\Delta T$ finito, reacción. Expansión libre ($V_2=2V_1$):
$$S_{gen}=mR\ln\frac{V_2}{V_1}=mR\ln2>0,$$
sin trabajo ni calor: puro potencial disipado.}
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- Trabajo reversible e irreversibilidad ----
\begin{frame}{Trabajo reversible e irreversibilidad}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\centering
\includegraphics[width=\linewidth,height=3.7cm,keepaspectratio]{reversibilidad}\\[0.5mm]
\fsub{El trabajo reversible es el máximo: $W_{rev}=\int P\,dv>W_{irr}$.}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardidea}{Exergía y estado muerto}{\faBalanceScale}
La \textbf{exergía} es el trabajo útil máximo que se obtiene al llevar el sistema reversiblemente al \textbf{estado muerto} $(T_0,P_0)$, en equilibrio con el entorno.
\end{cardidea}
\vspace{1.0mm}
\begin{cardgear}{Irreversibilidad (Gouy--Stodola)}{\faSuperscript}
{\footnotesize
La irreversibilidad es la exergía destruida:
$$I=\dot X_{dest}=T_0\,\dot S_{gen}\ge0.$$
Todo lo que genera entropía destruye trabajo útil a razón de $T_0$ por unidad de $S_{gen}$.}
\end{cardgear}
\end{column}
\end{columns}
\end{frame}

% ---- ¿Qué es la exergía? ----
\begin{frame}{¿Qué es la exergía? --- disponibilidad}
\begin{cardidea}{Definición}{\faInfoCircle}
La \textbf{exergía} (disponibilidad) es el \emph{trabajo útil máximo} extraíble de un sistema hasta el equilibrio con el entorno $(T_0,P_0)$. A diferencia de la energía (que se conserva), la exergía \textbf{se destruye} en todo proceso real: es la ``calidad'' de la energía.
\end{cardidea}
\vspace{1.2mm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{cardgear}{Sistema cerrado}{\faSuperscript}
{\footnotesize
$$B=(U-U_0)+P_0(V-V_0)-T_0(S-S_0).$$
$W_{\acute util,\max}=-\Delta B$ (reversible).\\[0.5mm]
Es la energía de Helmholtz si $V$ es constante.}
\end{cardgear}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardgear}{Flujo (sistema abierto)}{\faSuperscript}
{\footnotesize
$$\psi=(h-h_0)-T_0(s-s_0)+\tfrac{V^2}{2}+gz.$$
Usa $h$ porque el flujo ya pagó $Pv$.\\[0.5mm]
Agua a $T_0{=}25^\circ$C: $h_0{\approx}104.9$, $s_0{\approx}0.3674$.}
\end{cardgear}
\end{column}
\end{columns}
\vspace{1.0mm}
\begin{cardnota}[\faInfoCircle]
\textbf{Ejemplo de intuición:} 1~L de agua a 100~$^\circ$C y otro a 40~$^\circ$C tienen energía parecida, pero el primero tiene mucha más \emph{exergía} (mayor temperatura de fuente para un ciclo de potencia).
\end{cardnota}
\end{frame}

% ---- Gouy-Stodola / destrucción de exergía ----
\begin{frame}{Destrucción de exergía --- Gouy--Stodola}
\begin{columns}[T]
\begin{column}{0.45\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.1cm,keepaspectratio]{exergia_destruccion_Gouy_Stodola}\\[0.5mm]
\fsub{La exergía de entrada se reparte en útil, empuje al entorno y destruida.}
\end{column}
\begin{column}{0.51\textwidth}
\begin{cardidea}{Teorema de Gouy--Stodola}{\faBalanceScale}
Combinando 1.\textsuperscript{a} y 2.\textsuperscript{a} ley, el trabajo útil real es
$$W_{\acute util}=-\Delta B-T_0S_{gen},$$
de donde la \textbf{exergía destruida} es
$$\boxed{X_{dest}=T_0\,S_{gen}\ge0.}$$
\end{cardidea}
\vspace{1.0mm}
\begin{cardnota}[\faInfoCircle]
\textbf{Fuentes de destrucción:} calor con $\Delta T$ finito ($X_{dest}\uparrow$ con $\Delta T$), mezcla, fricción, estrangulamiento, reacción. El análisis exergético señala \emph{cuál} domina en cada equipo.
\end{cardnota}
\end{column}
\end{columns}
\end{frame}

% ---- Eficiencia de segunda ley ----
\begin{frame}{Eficiencia de segunda ley $\eta_{II}$ (exergética)}
\begin{cardidea}{Primera vs.\ segunda ley}{\faBalanceScale}
La eficiencia de \textbf{1.\textsuperscript{a} ley} $\eta$ compara \emph{energías} (cuánta se transfiere). La de \textbf{2.\textsuperscript{a} ley} $\eta_{II}=\epsilon$ compara \emph{exergías} (cuánto potencial de trabajo se aprovecha):
$$\eta_{II}=\epsilon=\frac{W_{\acute util}}{-\Delta B_{ent}}=1-\frac{X_{dest}}{-\Delta B_{ent}}=1-\frac{T_0S_{gen}}{-\Delta B_{ent}}.$$
\end{cardidea}
\vspace{1.4mm}
\begin{center}
{\footnotesize
\renewcommand{\arraystretch}{1.25}\arrayrulecolor{Granate}
\begin{tabular}{@{}>{\RaggedRight}p{5.4cm} >{\centering}p{2.3cm} >{\centering}p{2.3cm} >{\centering\arraybackslash}p{2.6cm}@{}}
\toprule
{\color{FondoOscuro}\bfseries Dispositivo} & {\color{FondoOscuro}\bfseries $\eta$ (1.\textsuperscript{a})} & {\color{FondoOscuro}\bfseries $\epsilon$ (2.\textsuperscript{a})} & {\color{FondoOscuro}\bfseries Causa}\\
\midrule
Calefactor eléctrico & $100\%$ & $<10\%$ & trabajo $\to$ calor frío\\
Turbina de vapor moderna & $88\%$ & $80\%$ & irreversibilidad interna\\
Intercambiador bien diseñado & $95\%$ & $60\%$ & $\Delta T$ grande\\
\bottomrule
\end{tabular}}
\end{center}
\vspace{0.8mm}
\begin{cardnota}[\faInfoCircle]
La eficiencia exergética revela dónde se pierde \emph{de verdad} el potencial de trabajo, no solo cuánta energía se transfiere: el calefactor es 100\% energético pero casi nulo en 2.\textsuperscript{a} ley.
\end{cardnota}
\end{frame}

% ---- Ejemplo numérico exergético ----
\begin{frame}{Ejemplo --- análisis exergético de una turbina}
\begin{cardgear}{Datos}{\faCalculator}
{\footnotesize
Vapor: entrada $P_1=6$ MPa, $T_1=500^\circ$C ($h_1=3422.2$, $s_1=6.882$); salida real $h_2=2330.2$, $s_2=7.470$ kJ/kg. Entorno $T_0=298.15$ K. Estado muerto del agua: $h_0=104.9$, $s_0=0.3674$.}
\end{cardgear}
\vspace{1.2mm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{cardidea}{Exergías de flujo}{\faSuperscript}
{\footnotesize
$\psi_1=(h_1-h_0)-T_0(s_1-s_0)$\\
$\phantom{\psi_1}=3317.3-1942.1=1375.2$.\\[0.6mm]
$\psi_2=(h_2-h_0)-T_0(s_2-s_0)$\\
$\phantom{\psi_2}=2225.3-2117.6=107.7$ kJ/kg.\\[0.6mm]
Trabajo real: $w=h_1-h_2=1092.0$ kJ/kg.}
\end{cardidea}
\end{column}
\begin{column}{0.49\textwidth}
\begin{cardidea}{Destrucción y eficiencia}{\faBalanceScale}
{\footnotesize
$X_{dest}=\psi_1-\psi_2-w=175.5$ kJ/kg.\\[0.6mm]
Chequeo Gouy--Stodola:\\
$T_0\,\Delta s_{gen}=298.15\times0.588=175.3$ kJ/kg.\;\checkmark\\[0.6mm]
$\displaystyle\eta_{II}=\frac{w}{\psi_1-\psi_2}=\frac{1092.0}{1267.5}=86.2\%$.}
\end{cardidea}
\end{column}
\end{columns}
\vspace{1.0mm}
\begin{cardnota}[\faInfoCircle]
$\eta_T=88\%$ (isentrópica) vs.\ $\epsilon=86.2\%$ (exergética): cercanas porque en una turbina adiabática la única irreversibilidad relevante es la generación interna de entropía, $X_{dest}=T_0S_{gen}$.
\end{cardnota}
\end{frame}
