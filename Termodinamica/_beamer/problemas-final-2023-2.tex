% ============================================================
%  PROBLEMA 1 — CICLO OTTO
% ============================================================
\portadaseccion{1}{Problema 1 --- Ciclo Otto}{Motor de 1,6 L, 4 cilindros, 76 kW}{\faCalculator}

% ---- P1: Enunciado y datos ----
\begin{frame}{P1 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Motor \textbf{Otto ideal} de $1{,}6$ L de cilindrada, \textbf{4 tiempos y 4 cilindros}, con relación de compresión $r_k=6{,}2$, que entrega una potencia $\dot W=76{,}1$ kW. Coeficiente adiabático $k=1{,}4$.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} $V_{cil}=1{,}6$ L $=1600$ cm$^3$ (total) · $N=4$ cil\\[0.5mm]
$r_k=6{,}2$ · $k=1{,}4$ · $\dot W=76{,}1$ kW\\[0.5mm]
Combustión isocórica ($2\to3$), 4 tiempos\\[0.5mm]
\textbf{Pide:} (1)~$V_2$ muerto [cm$^3$] · (2)~$\eta$ · (3)~$\dot Q_H$ · (4)~$\dot Q_L$
\end{cardlista}
\end{column}
\begin{column}{0.43\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.4cm,keepaspectratio]{otto_diagrama_Pv}\\[0.4mm]
\fsub{Diagrama $P$-$v$: combustión isocórica ($2\to3$), rechazo isocórico ($4\to1$).}
\end{column}
\end{columns}
\end{frame}

% ---- P1: Solución ----
\begin{frame}{P1 --- Solución paso a paso}
\begin{cardgear}{Paso 1 --- (1) Volumen muerto por cilindro}{\faSuperscript}
Cilindrada por cilindro $=1600/4=400$ cm$^3=V_1-V_2$, con $V_1/V_2=r_k=6{,}2$:
$$V_2=\frac{V_1-V_2}{r_k-1}=\frac{400}{6{,}2-1}=\frac{400}{5{,}2}=76{,}9\ \text{cm}^3.$$
\end{cardgear}
\vspace{1.1mm}
\begin{cardidea}{Paso 2 --- (2) Rendimiento del ciclo}{\faListOl}
Depende solo de $r_k$:\quad $\eta=1-\dfrac{1}{r_k^{\,k-1}}=1-(6{,}2)^{-0{,}4}=51{,}8\%.$
\end{cardidea}
\vspace{1.1mm}
\begin{cardgear}{Paso 3 --- (3) y (4) Calor absorbido y rechazado}{\faFire}
De $\eta=\dot W/\dot Q_H$ y del balance $\dot Q_L=\dot Q_H-\dot W$:
$$\dot Q_H=\frac{\dot W}{\eta}=\frac{76{,}1}{0{,}518}=146{,}9\ \text{kW},\qquad \dot Q_L=146{,}9-76{,}1=70{,}8\ \text{kW}.$$
\end{cardgear}
\vspace{1.1mm}
\begin{cardnota}[\faListOl]
\textbf{Resultado:} $V_2=76{,}9$ cm$^3$ · $\eta=51{,}8\%$ · $\dot Q_H=146{,}9$ kW · $\dot Q_L=70{,}8$ kW.
\end{cardnota}
\end{frame}

% ============================================================
%  PROBLEMA 2 — TURBORREACTOR (TURBOJET)
% ============================================================
\portadaseccion{2}{Problema 2 --- Turborreactor}{Avión turbo jet, propulsión a chorro}{\faCalculator}

% ---- P2: Enunciado y datos ----
\begin{frame}{P2 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Avión \textbf{turbo jet} que vuela a $C_a=280$ m/s. El aire entra al \textbf{difusor} a $-10\,^\circ$C y $80$ kPa; el \textbf{compresor} tiene $r_p=10$; la temperatura máxima es $1100\,^\circ$C. La \textbf{turbina} da el trabajo justo para mover el compresor y la \textbf{tobera} expande (adiabática reversible) hasta $80$ kPa. Flujo $\dot m=60$ kg/s.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} $C_a=280$ m/s · difusor: $-10\,^\circ$C, $80$ kPa\\[0.5mm]
$r_p=10$ · $T_{max}=T_3=1100\,^\circ$C · $\dot m=60$ kg/s\\[0.5mm]
$w_T=w_C$ · tobera hasta $80$ kPa · aire real (CATT3)\\[0.5mm]
\textbf{Pide:} (5)~$T_4,P_4$ · (6)~$T_2,P_2$ · (7)~$C_5$ · (8)~$\dot W_p$ · (9)~$\eta_p$
\end{cardlista}
\end{column}
\begin{column}{0.43\textwidth}
\centering
\includegraphics[width=\linewidth,height=3.7cm,keepaspectratio]{brayton_esquema_dispositivos}\\[0.6mm]
\fsub{Núcleo del turborreactor: compresor $1\to2$, cámara $2\to3$, turbina $3\to4$ ($w_T=w_C$); un difusor antecede y una tobera sigue.}
\end{column}
\end{columns}
\end{frame}

% ---- P2: Solución (difusor y compresor) ----
\begin{frame}{P2 --- Solución paso a paso (difusor y compresor)}
\begin{cardnota}[\faTable]
{\scriptsize\textbf{Estados (aire real, CATT3):}\ \ $a$: $h{=}263{,}5$, $P{=}80$\ $\vert$\ 1: $h{=}302{,}7$, $P{=}129{,}84$\ $\vert$\ 2: $h{=}584{,}7$, $P{=}1298{,}5$\ $\vert$\ 3: $h{=}1483$, $P{=}1298{,}5$\ $\vert$\ 4: $h{=}1201$, $P{=}590{,}8$\ $\vert$\ 5: $h{=}690{,}1$, $P{=}80$\ \ ($h$ en kJ/kg, $P$ en kPa).}
\end{cardnota}
\vspace{1.2mm}
\begin{cardgear}{Paso 1 --- Difusor ($a\to1$): entalpía de estancamiento}{\faSuperscript}
El difusor frena el aire; se conserva $h_0=h+\tfrac{C^2}{2}$ (con $C_1\approx0$):
$$h_1=h_a+\tfrac{C_a^2}{2}=263{,}5+\tfrac{280^2}{2}\times10^{-3}=302{,}7\ \text{kJ/kg}\ \Rightarrow\ s_1=6{,}877.$$
$$s_1=s_a+R\ln\!\tfrac{P_1}{P_a}\ \Rightarrow\ P_1=129{,}84\ \text{kPa}.$$
\end{cardgear}
\vspace{1.1mm}
\begin{cardidea}{Paso 2 --- (6) Estado 2 (salida del compresor)}{\faListOl}
Compresor con $r_p=10$: $P_2=10\,P_1=1298{,}5$ kPa. Isentrópico ($s_2=s_1$):
$$h_2=584{,}7\ \text{kJ/kg}\ \Rightarrow\ T_2=305{,}2\,^\circ\text{C}.$$
\end{cardidea}
\end{frame}

% ---- P2: Solución (turbina y tobera) ----
\begin{frame}{P2 --- Solución paso a paso (turbina y tobera)}
\begin{cardgear}{Paso 3 --- (5) Estado 4 (salida de la turbina)}{\faSuperscript}
La turbina entrega exactamente el trabajo del compresor: $h_3-h_4=h_2-h_1$:
$$h_4=1483-(584{,}7-302{,}7)=1201\ \text{kJ/kg}\ \Rightarrow\ T_4=861{,}2\,^\circ\text{C}.$$
Por la isentrópica desde el estado 3 se obtiene $P_4=590{,}8$ kPa.
\end{cardgear}
\vspace{1.2mm}
\begin{cardidea}{Paso 4 --- (7) Velocidad de salida (tobera $4\to5$)}{\faListOl}
Expansión isentrópica hasta $P_5=80$ kPa ($s_5=s_4$): $h_5=690{,}1$ kJ/kg. Con $h_4=h_5+\tfrac{C_5^2}{2}$:
$$C_5=\sqrt{2(h_4-h_5)\times10^3}=\sqrt{2(1201-690{,}1)\times10^3}=1010\ \text{m/s}.$$
\end{cardidea}
\end{frame}

% ---- P2: Solución (potencia y eficiencia) ----
\begin{frame}{P2 --- Solución paso a paso (potencia y $\eta_p$)}
\begin{cardidea}{Paso 5 --- (8) Potencia de propulsión}{\faListOl}
Empuje $F=\dot m(C_5-C_a)$; potencia de propulsión $\dot W_p=F\,C_a$:
$$\dot W_p=\dot m(C_5-C_a)\,C_a=60(1010-280)(280)\times10^{-3}=12\,279\ \text{kW}.$$
\end{cardidea}
\vspace{1.4mm}
\begin{cardgear}{Paso 6 --- (9) Eficiencia de propulsión}{\faFire}
Calor aportado en la cámara ($2\to3$): $\dot Q_H=\dot m(h_3-h_2)=60(1483-584{,}7)=53\,898$ kW:
$$\eta_p=\frac{\dot W_p}{\dot Q_H}=\frac{12\,279}{53\,898}=22{,}78\%.$$
\end{cardgear}
\vspace{1.4mm}
\begin{cardnota}[\faListOl]
\textbf{Resultado:} $T_4=861{,}2\,^\circ$C, $P_4=590{,}8$ kPa · $T_2=305{,}2\,^\circ$C, $P_2=1298{,}5$ kPa · $C_5=1010$ m/s · $\dot W_p=12\,279$ kW · $\eta_p=22{,}78\%$.
\end{cardnota}
\end{frame}

% ---- P3: Enunciado y datos ----
\begin{frame}{P3 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Motor teórico de \textbf{encendido por compresión} (Diesel), relación aire-combustible $r_{a/c}=16$, $PC=36\,900$ kJ/kg. Al final de la expansión el gas ocupa $5$ L por cilindro a $1545$ K. Es de \textbf{4 tiempos, 4 cilindros}, gira a $840$ rpm; al inicio de la compresión $T_1=300$ K y $P_1=100$ kPa.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} $r_{a/c}=16$ · $PC=36\,900$ kJ/kg\\[0.5mm]
$V_4=V_1=5$ L $=5000$ cm$^3$ · $T_4=1545$ K\\[0.5mm]
$N=4$ cil · $840$ rpm · $T_1=300$ K · $P_1=100$ kPa\\[0.5mm]
$k=1{,}4$ · $c_p=1{,}0035$ · $R=0{,}287$ kJ/kg·K\\[0.5mm]
\textbf{Pide:} (10)~$r_c$ · (11)~$V_2$ · (12)~$\dot m_{gas}$ · (13)~$\dot m_c$ · (14)~$\dot W$
\end{cardlista}
\end{column}
\begin{column}{0.43\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.4cm,keepaspectratio]{diesel_diagrama_Pv}\\[0.4mm]
\fsub{Diagrama $P$-$v$: combustión isobárica ($2\to3$), relación de corte $r_c$.}
\end{column}
\end{columns}
\end{frame}

% ---- P3: Solución (r_c, V_2) ----
\begin{frame}{P3 --- Solución paso a paso ($r_c$, $V_2$)}
\begin{cardgear}{Paso 1 --- (10) Relación de corte}{\faSuperscript}
El estado 4 (fin de expansión) cae sobre la isócora de admisión ($V_4=V_1$); para el Diesel aire-estándar $T_4/T_1=r_c^{\,k}$:
$$r_c=\left(\frac{T_4}{T_1}\right)^{1/k}=\left(\frac{1545}{300}\right)^{1/1{,}4}=3{,}224.$$
\end{cardgear}
\vspace{1.1mm}
\begin{cardidea}{Paso 2 --- (11) Volumen muerto por cilindro}{\faSuperscript}
{\footnotesize
La combustión isobárica ($\dot m_c\,PC=\dot m_a\,c_p(T_3-T_2)$, con $r_{a/c}=16$ y $T_3=r_c T_2$) fija $T_2$:
$$\frac{PC}{16}=c_p\,T_2\,(r_c-1)\ \Rightarrow\ T_2=\frac{36\,900/16}{1{,}0035(3{,}224-1)}=1033\ \text{K}.$$
Compresión isentrópica $T_2=T_1\,r_k^{\,k-1}\Rightarrow r_k=(T_2/T_1)^{1/(k-1)}=(3{,}444)^{2{,}5}=22{,}0$. Con $V_1=5000$ cm$^3$:
$$V_2=\frac{V_1}{r_k}=\frac{5000}{22{,}0}=227\ \text{cm}^3.$$}
\end{cardidea}
\end{frame}

% ---- P3: Solución (flujos y potencia) ----
\begin{frame}{P3 --- Solución paso a paso (flujos y potencia)}
\begin{cardgear}{Paso 3 --- (12) y (13) Flujos de masa}{\faListOl}
{\footnotesize
Masa de aire por cilindro y ciclo, y ciclos por minuto (4 tiempos: $\tfrac{840}{2}\times4$ cil):
$$m_a=\frac{P_1V_1}{R\,T_1}=\frac{100(5000\times10^{-6})}{0{,}287(300)}=5{,}81\times10^{-3}\ \text{kg},\quad n_{ciclos}=\tfrac{840}{2}\times4=1680\ \text{/min}.$$
$$\dot m_{gas}=\big(m_a+\tfrac{m_a}{16}\big)\,n_{ciclos}=6{,}17\times10^{-3}(1680)=10{,}4\ \tfrac{\text{kg}}{\text{min}},\ \ \dot m_c=\tfrac{m_a}{16}\,n_{ciclos}\,60=36{,}6\ \tfrac{\text{kg}}{\text{h}}.$$}
\end{cardgear}
\vspace{0.9mm}
\begin{cardidea}{Paso 4 --- (14) Eficiencia y potencia teórica}{\faFire}
{\footnotesize
$$\eta_D=1-\frac{1}{r_k^{\,k-1}}\frac{r_c^{\,k}-1}{k(r_c-1)}=1-\frac{1}{22^{0{,}4}}\cdot\frac{3{,}224^{1{,}4}-1}{1{,}4(3{,}224-1)}=61{,}6\%.$$
Con $Q_H$ por ciclo $=m_c\,PC=3{,}63\times10^{-4}(36\,900)=13{,}4$ kJ:
$$\dot W=\eta_D\,Q_H\cdot\frac{n_{ciclos}}{60}=0{,}616(13{,}4)(28)=231\ \text{kW}.$$}
\end{cardidea}
\vspace{0.7mm}
\begin{cardnota}[\faListOl]
\textbf{Resultado:} $r_c=3{,}224$ · $V_2=227$ cm$^3$ · $\dot m_{gas}=10{,}4$ kg/min · $\dot m_c=36{,}6$ kg/h · $\dot W=231$ kW.
\end{cardnota}
\end{frame}

\portadaseccion{4}{Problema 4 --- Ciclo Rankine}{Planta térmica a vapor, 14 bar, 20 MW}{\faCalculator}

% ---- P4: Enunciado y datos ----
\begin{frame}{P4 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Planta térmica a vapor (\textbf{Rankine simple}). Presión en el caldero $14$ bar; la temperatura del vapor en el condensador es $60\,^\circ$C y la calidad a la salida de la turbina es $x_4=95{,}66\%$. Se desprecia el trabajo de la bomba. La turbina desarrolla $20$ MW.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} $P_1=14$ bar · $T_{cond}=60\,^\circ$C · $x_4=0{,}9566$\\[0.5mm]
$\dot W_T=20$ MW · $w_B\approx0$ · $PC=40\,000$ kJ/kg\\[0.5mm]
Agua refrig.: $\dot m_w=1200$ kg/s, $c_p=4{,}18$, $T_{in}=20\,^\circ$C\\[0.5mm]
\textbf{Pide:} (15)~$T_{max}$ · (16)~$P_{cond}$ · (17)~$\eta$ · (18)~$\dot m_v$ · (19)~$\dot m_c$ · (20)~$T_{out}$
\end{cardlista}
\end{column}
\begin{column}{0.44\textwidth}
\centering
\includegraphics[width=\linewidth,height=2.9cm,keepaspectratio]{rankine_esquema_planta}\\[0.6mm]
\includegraphics[width=\linewidth,height=2.7cm,keepaspectratio]{rankine_diagrama_Ts}\\[0.3mm]
\fsub{Esquema de la planta y ciclo en el diagrama $T$-$s$.}
\end{column}
\end{columns}
\end{frame}

% ---- P4: Solución (estados) ----
\begin{frame}{P4 --- Solución paso a paso (estados y eficiencia)}
\begin{cardgear}{Paso 1 --- (15) y (16) Estados del ciclo}{\faSuperscript}
{\footnotesize
\textbf{Salida de turbina} ($60\,^\circ$C, $x_4=0{,}9566$):
$$s_4=s_f+x_4\,s_{fg}=0{,}8313+0{,}9566(7{,}0784)=7{,}602\ \tfrac{\text{kJ}}{\text{kg·K}},\ \ h_4=251{,}1+0{,}9566(2358{,}5)=2507\ \tfrac{\text{kJ}}{\text{kg}}.$$
Turbina isentrópica: $s_1=s_4=7{,}602$ a $P_1=14$ bar $\Rightarrow$ vapor sobrecalentado, $T_{max}\approx500\,^\circ$C ($h_1=3474$). Condensador: $P_{cond}=P_{sat}(60\,^\circ\text{C})=19{,}94$ kPa.}
\end{cardgear}
\vspace{1mm}
\begin{cardidea}{Paso 2 --- (17)--(19) Eficiencia y flujos}{\faListOl}
{\footnotesize
Despreciando la bomba ($h_3\approx h_f(60\,^\circ\text{C})=251{,}1$):
$$\eta=\frac{h_1-h_4}{h_1-h_3}=\frac{3474-2507}{3474-251{,}1}=30\%.$$
$$\dot m_v=\frac{\dot W_T}{h_1-h_4}=\frac{20\,000}{967}=20{,}68\ \tfrac{\text{kg}}{\text{s}},\quad \dot m_c=\frac{\dot m_v(h_1-h_3)}{PC}=\frac{20{,}68(3222{,}9)}{40\,000}=1{,}666\ \tfrac{\text{kg}}{\text{s}}.$$}
\end{cardidea}
\end{frame}

% ---- P4: Solución (agua de refrigeración) ----
\begin{frame}{P4 --- Solución paso a paso (agua de refrigeración)}
\begin{cardgear}{Paso 3 --- (20) Temperatura de salida del agua}{\faSnowflake}
Balance de energía en el condensador, $\dot m_v(h_4-h_3)=\dot m_w\,c_p\,(T_{out}-T_{in})$:
$$20{,}68(2507-251{,}1)=1200(4{,}18)(T_{out}-20),$$
$$46\,650=5016\,(T_{out}-20)\ \Rightarrow\ T_{out}=20+9{,}3=29{,}3\,^\circ\text{C}.$$
\end{cardgear}
\vspace{1.2mm}
\begin{cardnota}[\faExclamationTriangle]
{\footnotesize \textbf{Nota (discrepancia con la clave):} los incisos 15--19 coinciden con la clave. En el (20) la clave reporta $46{,}27\,^\circ$C, pero con $\dot m_w=1200$ kg/s el balance da $\Delta T=46\,650/(1200\cdot4{,}18)=9{,}3\,^\circ$C, es decir $T_{out}=29{,}3\,^\circ$C (el valor de la clave requeriría $\dot m_w\approx425$ kg/s).}
\end{cardnota}
\vspace{1.2mm}
\begin{cardnota}[\faListOl]
\textbf{Resultado:} $T_{max}\approx500\,^\circ$C · $P_{cond}=19{,}94$ kPa · $\eta=30\%$ · $\dot m_v=20{,}68$ kg/s · $\dot m_c=1{,}666$ kg/s · $T_{out}=29{,}3\,^\circ$C.
\end{cardnota}
\end{frame}
