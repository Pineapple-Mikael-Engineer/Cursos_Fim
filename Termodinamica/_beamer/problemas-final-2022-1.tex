%  PROBLEMA 1 — CICLO OTTO
% ============================================================
\portadaseccion{1}{Problema 1 --- Ciclo Otto}{Presión media indicada, $r=8$}{\faCalculator}

% ---- P1: Enunciado y datos ----
\begin{frame}{P1 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
En un ciclo \textbf{Otto}, la compresión adiabática arranca a $108$ kPa y $303$ K, la relación de compresión es $r=8$ y el calor agregado es $q_{in}=2250$ kJ/kg. Determinar el diagrama $P$-$v$ con presiones y temperaturas, la \textbf{presión media indicada} y la eficiencia térmica.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} $P_1=108$ kPa · $T_1=303$ K · $r=8$\\[0.5mm]
$q_{in}=q_H=2250$ kJ/kg · $k=1{,}4$\\[0.5mm]
$c_v=0{,}718$ · $R=0{,}287$ kJ/kg·K\\[0.5mm]
\textbf{Pide:} (a)~estados $P,T$ · (b)~PMI · (c)~$\eta_{th}$
\end{cardlista}
\end{column}
\begin{column}{0.43\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.6cm,keepaspectratio]{otto_diagrama_Pv}\\[0.4mm]
\fsub{Diagrama $P$-$v$: adición de calor isocórica ($2\to3$).}
\end{column}
\end{columns}
\end{frame}

% ---- P1: Solución (estados) ----
\begin{frame}{P1 --- Solución paso a paso (estados)}
\begin{cardgear}{Paso 1 --- Compresión isentrópica $1\to2$}{\faSuperscript}
$$T_2=T_1\,r^{\,k-1}=303\,(8)^{0{,}4}=696{,}1\ \text{K},\quad P_2=P_1\,r^{\,k}=108\,(8)^{1{,}4}=1985\ \text{kPa}.$$
\end{cardgear}
\vspace{1.2mm}
\begin{cardidea}{Paso 2 --- Adición isocórica de calor $2\to3$}{\faFire}
El calor entra a volumen constante; $\ q_H=c_v(T_3-T_2)$:
$$T_3=T_2+\frac{q_{in}}{c_v}=696{,}1+\frac{2250}{0{,}718}=3829{,}8\ \text{K},\quad P_3=P_2\frac{T_3}{T_2}=1985\cdot\frac{3829{,}8}{696{,}1}=10\,921\ \text{kPa}.$$
\end{cardidea}
\vspace{1.2mm}
\begin{cardgear}{Paso 3 --- Expansión isentrópica $3\to4$}{\faSuperscript}
$$T_4=\frac{T_3}{r^{\,k-1}}=\frac{3829{,}8}{(8)^{0{,}4}}=1667\ \text{K},\quad P_4=P_1\frac{T_4}{T_1}=108\cdot\frac{1667}{303}=594\ \text{kPa}.$$
\end{cardgear}
\end{frame}

% ---- P1: Solución (PMI y eficiencia) ----
\begin{frame}{P1 --- Solución paso a paso (PMI y $\eta$)}
\begin{cardgear}{Paso 4 --- (c) Eficiencia térmica y trabajo neto}{\faListOl}
La eficiencia del Otto depende sólo de $r$:
$$\eta_{th}=1-\frac{1}{r^{\,k-1}}=1-(8)^{-0{,}4}=56{,}5\%,\qquad w_{neto}=\eta_{th}\,q_{in}=0{,}565\cdot2250=1271\ \tfrac{\text{kJ}}{\text{kg}}.$$
\end{cardgear}
\vspace{1.4mm}
\begin{cardidea}{Paso 5 --- (b) Presión media indicada}{\faListOl}
Volúmenes específicos extremos y $\text{PMI}=w_{neto}/(v_1-v_2)$:
$$v_1=\frac{R\,T_1}{P_1}=\frac{0{,}287(303)}{108}=0{,}805\ \tfrac{\text{m}^3}{\text{kg}},\quad v_2=\frac{v_1}{r}=0{,}1007\ \tfrac{\text{m}^3}{\text{kg}},$$
$$\text{PMI}=\frac{w_{neto}}{v_1-v_2}=\frac{1271}{0{,}705}=1803\ \text{kPa}.$$
\end{cardidea}
\vspace{0.8mm}
\begin{cardnota}[\faListOl]
\textbf{Resultado:} $T_2=696{,}1$ K · $T_3=3829{,}8$ K · $\eta_{th}=56{,}5\%$ · $w_{neto}=1271$ kJ/kg · $\text{PMI}=1803$ kPa.
\end{cardnota}
\end{frame}

\portadaseccion{2}{Problema 2 --- Brayton ideal}{Turbina a gas Joule-Brayton estándar}{\faCalculator}

% ---- P2: Enunciado y datos ----
\begin{frame}{P2 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Planta de potencia con turbina a gas en ciclo \textbf{Joule-Brayton estándar}. El aire entra al compresor a $312$ K y $107$ kPa, la relación de presiones es $r_p=11$ y la temperatura máxima es $1420$ K. Hallar $P,T$ en cada estado, los trabajos de compresor y turbina, y la eficiencia térmica.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} $T_1=312$ K · $P_1=107$ kPa · $r_p=11$\\[0.5mm]
$T_3=T_{max}=1420$ K · $k=1{,}4$ · $c_p=1{,}005$\\[0.5mm]
$\tfrac{k-1}{k}=0{,}2857$\\[0.5mm]
\textbf{Pide:} (a)~estados · (b)~$w_C,w_T,\eta_{th}$
\end{cardlista}
\end{column}
\begin{column}{0.44\textwidth}
\centering
\includegraphics[width=\linewidth,height=3.0cm,keepaspectratio]{brayton_diagrama_Ts}\\[0.6mm]

\fsub{Diagrama $T$-$s$ y esquema compresor--cámara--turbina.}
\end{column}
\end{columns}
\end{frame}

% ---- P2: Solución ----
\begin{frame}{P2 --- Solución paso a paso}
\begin{cardgear}{Paso 1 --- Estados isentrópicos}{\faSuperscript}
{\footnotesize
\textbf{Compresor} ($1\to2$): $T_2=T_1\,r_p^{\,0{,}2857}=312\,(11)^{0{,}2857}=619{,}3$ K,\ \ $P_2=r_p P_1=1177$ kPa.\\[0.8mm]
\textbf{Turbina} ($3\to4$): $T_3=1420$ K, $P_3=1177$ kPa;\ \ $T_4=\dfrac{T_3}{r_p^{\,0{,}2857}}=\dfrac{1420}{(11)^{0{,}2857}}=715{,}4$ K, $P_4=107$ kPa.}
\end{cardgear}
\vspace{1.2mm}
\begin{cardidea}{Paso 2 --- (b) Trabajos específicos}{\faListOl}
$$w_C=c_p(T_2-T_1)=1{,}005(619{,}3-312)=308{,}8\ \tfrac{\text{kJ}}{\text{kg}},$$
$$w_T=c_p(T_3-T_4)=1{,}005(1420-715{,}4)=708{,}1\ \tfrac{\text{kJ}}{\text{kg}},\quad w_{neto}=w_T-w_C=399{,}3\ \tfrac{\text{kJ}}{\text{kg}}.$$
\end{cardidea}
\vspace{1.2mm}
\begin{cardgear}{Paso 3 --- (b) Eficiencia térmica}{\faListOl}
Depende sólo de $r_p$:\quad $\eta_{th}=1-\dfrac{1}{r_p^{\,0{,}2857}}=1-\dfrac{1}{(11)^{0{,}2857}}=49{,}6\%.$
\end{cardgear}
\vspace{0.8mm}
\begin{cardnota}[\faListOl]
\textbf{Resultado:} $T_2=619{,}3$ K · $T_4=715{,}4$ K · $w_C=308{,}8$ · $w_T=708{,}1$ · $w_{neto}=399{,}3$ kJ/kg · $\eta_{th}=49{,}6\%$.
\end{cardnota}
\end{frame}

\portadaseccion{3}{Problema 3 --- Rankine}{Eficiencia y temperatura media de adición}{\faCalculator}

% ---- P3: Enunciado y datos ----
\begin{frame}{P3 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Ciclo \textbf{Rankine}: el vapor entra a la turbina a $30$ bar y el condensador está a $0{,}1$ bar. \textbf{(a)} Eficiencia y \textbf{temperatura media de adición} de calor con vapor \emph{saturado}. \textbf{(b)} Si hay sobrecalentamiento hasta $600\,^\circ$C, hallar $\eta$ y $T_m$.
\end{cardidea}
\vspace{1mm}
\begin{cardgear}{Temperatura media de adición}{\faSuperscript}
{\footnotesize El ciclo equivale a un Carnot entre $T_m$ y $T_{cond}$:
$$T_m=\frac{q_{ent}}{\Delta s_{ent}}=\frac{h_1-h_4}{s_1-s_4}.$$
Subir $T_m$ mejora $\eta$.}
\end{cardgear}
\end{column}
\begin{column}{0.43\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.6cm,keepaspectratio]{rankine_diagrama_Ts}\\[0.4mm]
\fsub{Diagrama $T$-$s$: adición de calor en la caldera.}
\end{column}
\end{columns}
\end{frame}

% ---- P3: Solución (a) ----
\begin{frame}{P3 --- Solución (a): vapor saturado a $30$ bar}
\begin{cardgear}{Paso 1 --- Estados}{\faSuperscript}
{\footnotesize
\textbf{Estado 1} (vapor sat. a $3$ MPa): $h_1=2804$, $s_1=6{,}187$. \textbf{Turbina isentrópica} a $10$ kPa ($s_f=0{,}6493$, $s_{fg}=7{,}5009$):
$$x_2=\frac{6{,}187-0{,}6493}{7{,}5009}=0{,}738,\quad h_2=191{,}8+0{,}738(2392{,}8)=1958\ \tfrac{\text{kJ}}{\text{kg}}.$$
\textbf{Bomba}: $h_4=h_3+v\,\Delta P=191{,}8+0{,}001(3000-10)=194{,}8\ \tfrac{\text{kJ}}{\text{kg}}.$}
\end{cardgear}
\vspace{1.2mm}
\begin{cardidea}{Paso 2 --- (a) Eficiencia y temperatura media}{\faListOl}
$$\eta=\frac{(h_1-h_2)-w_B}{h_1-h_4}=\frac{846-3}{2609}=32{,}3\%,$$
$$T_m=\frac{h_1-h_4}{s_1-s_4}=\frac{2609}{6{,}187-0{,}649}=471\ \text{K}=198\,^\circ\text{C}.$$
\end{cardidea}
\end{frame}

\begin{frame}{P3 --- Solución (b): sobrecalentado a $600\,^\circ$C}

$$x_2=\frac{7{,}510-0{,}6493}{7{,}5009}=0{,}915,\quad h_2=191{,}8+0{,}915(2392{,}8)=2381\ \tfrac{\text{kJ}}{\text{kg}}.$$}
\end{cardgear}

\begin{cardidea}{Paso 4 --- (b) Eficiencia y temperatura media}{\faListOl}

\end{cardidea}
\vspace{1mm}

El sobrecalentamiento sube $T_m$ de $198$ a $235\,^\circ$C y con ello $\eta$ de $32{,}3$ a $37{,}2\%$, además de mejorar la calidad de salida ($0{,}74\to0{,}92$).

\end{frame}

\begin{cardgear}{Paso 1 --- Estados de entrada y salida}{\faSuperscript}
{\footnotesize
\textbf{Entrada} ($4{,}4$ MPa, $385\,^\circ$C, interpolado): $h_1\approx3169$ kJ/kg, $s_1\approx6{,}663$ kJ/kg·K.\\[0.8mm]

\end{cardidea}

\portadaseccion{6}{Problema 6 --- Exergía}{Turbina de vapor: análisis de 2ª ley}{\faCalculator}

% ---- P6: Enunciado y datos ----
\begin{frame}{P6 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Vapor entra a una \textbf{turbina} en régimen estacionario a $3{,}5$ MPa y $450\,^\circ$C con $\dot m=10$ kg/s, y sale a $0{,}22$ MPa y $170\,^\circ$C. La turbina pierde $\dot Q_{perd}=300$ kW hacia el aire a $100$ kPa y $25\,^\circ$C. $\Delta$EC, $\Delta$EP despreciables.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} $h_1=3337$, $s_1=7{,}005$\\[0.5mm]
$h_2=2810$, $s_2=7{,}370$ · estado muerto $h_0=104{,}8$, $s_0=0{,}367$\\[0.5mm]
$T_0=298{,}15$ K · $\dot m=10$ kg/s · $\dot Q_{perd}=300$ kW\\[0.5mm]
\textbf{Pide:} (a)~$\dot W_{real}$ · (b)~$\dot W_{max}$ · (c)~$\eta_{II}$ · (d)~$\dot X_{dest}$ · (e)~$\dot X_1$
\end{cardlista}
\end{column}
\begin{column}{0.43\textwidth}
\centering
\includegraphics[width=\linewidth,height=4.6cm,keepaspectratio]{rankine_esquema_planta}\\[0.4mm]
\fsub{Turbina con pérdida de calor al ambiente.}
\end{column}
\end{columns}
\end{frame}

% ---- P6: Solución (a,b,c) ----
\begin{frame}{P6 --- Solución paso a paso (potencia real y reversible)}
\begin{cardidea}{Paso 1 --- (a) Potencia real}{\faListOl}
Balance de energía con pérdida de calor:
$$\dot W_{real}=\dot m(h_1-h_2)-\dot Q_{perd}=10(3337-2810)-300=5270-300=4970\ \text{kW}.$$
\end{cardidea}
\vspace{1.2mm}
\begin{cardgear}{Paso 2 --- (b) Potencia máxima (reversible)}{\faSuperscript}
La pérdida de calor sale al ambiente a $T_0$ (no destruye exergía adicional):
$$\dot W_{max}=\dot m\big[(h_1-h_2)-T_0(s_1-s_2)\big]=10\big[527-298{,}15(7{,}005-7{,}370)\big]=6358\ \text{kW}.$$
\end{cardgear}
\vspace{1.2mm}
\begin{cardidea}{Paso 3 --- (c) Eficiencia de 2ª ley}{\faListOl}
$$\eta_{II}=\frac{\dot W_{real}}{\dot W_{max}}=\frac{4970}{6358}=78{,}2\%.$$
\end{cardidea}
\end{frame}

% ---- P6: Solución (d,e) ----
\begin{frame}{P6 --- Solución paso a paso (exergía destruida y de entrada)}
\begin{cardgear}{Paso 4 --- (d) Exergía destruida}{\faListOl}
$$\dot X_{dest}=\dot W_{max}-\dot W_{real}=6358-4970=1388\ \text{kW}\quad(=T_0\,\dot S_{gen}).$$
\end{cardgear}
\vspace{1.2mm}
\begin{cardidea}{Paso 5 --- (e) Exergía del vapor a la entrada}{\faSuperscript}
Exergía de flujo respecto del estado muerto ($25\,^\circ$C):
$$\dot X_1=\dot m\big[(h_1-h_0)-T_0(s_1-s_0)\big]=10\big[(3337-104{,}8)-298{,}15(7{,}005-0{,}367)\big]=12\,531\ \text{kW}.$$
\end{cardidea}
\vspace{1mm}
\begin{cardnota}[\faExclamationTriangle]
{\footnotesize \textbf{Discrepancia con la clave:} la nota redondea sus valores de vapor interpolados y reporta $\dot W_{real}=4977$, $\dot W_{max}=6366$, $\dot X_{dest}=1389$, $\dot X_1=12\,535$ kW. La aritmética exacta con los $h,s$ dados da $4970$, $6358$, $1388$ y $12\,531$ kW (diferencias $<8$ kW por redondeo); $\eta_{II}=78{,}2\%$ coincide.}
\end{cardnota}
\end{frame}

% ============================================================
\portadaseccion{9}{Problema 9 --- Cascada R-12}{Refrigeración en dos etapas}{\faCalculator}

% ---- P9: Enunciado y datos ----
\begin{frame}{P9 --- Enunciado y datos}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{cardidea}{Enunciado}{\faInfoCircle}
Sistema de \textbf{refrigeración en cascada} con \textbf{Freón-12}. El evaporador del ciclo inferior está a $-30\,^\circ$C con capacidad $25$ TON; un intercambiador de cascada acopla el condensador inferior ($\approx-10\,^\circ$C) con el evaporador superior ($\approx-15\,^\circ$C); el condensador superior está a $40\,^\circ$C. Hallar los flujos másicos, la potencia de entrada y el COP.
\end{cardidea}
\vspace{1mm}
\begin{cardlista}
\textbf{Datos:} R-12 · $\dot Q_L=25$ TON$=87{,}9$ kW\\[0.5mm]
Ciclo B (inferior): evap $-30\,^\circ$C, cond $-10\,^\circ$C\\[0.5mm]
Ciclo A (superior): evap $-15\,^\circ$C, cond $40\,^\circ$C\\[0.5mm]
\textbf{Pide:} (a)~$\dot m_A,\dot m_B$ · (b)~$\dot W$ · (c)~$\mathrm{COP}$
\end{cardlista}
\end{column}
\begin{column}{0.44\textwidth}
\centering
\includegraphics[width=\linewidth,height=3.0cm,keepaspectratio]{VCR_esquema_dispositivos}\\[0.6mm]
\includegraphics[width=\linewidth,height=2.4cm,keepaspectratio]{VCR_diagrama_Ph}\\[0.3mm]
\fsub{Cada etapa: compresión de vapor (esquema y $P$-$h$).}
\end{column}
\end{columns}
\end{frame}

% ---- P9: Solución (método) ----
\begin{frame}{P9 --- Solución paso a paso (acoplamiento y flujos)}
\begin{cardgear}{Paso 1 --- Acoplamiento de la cascada}{\faSnowflake}
El calor que \textbf{cede} el condensador inferior lo \textbf{absorbe} el evaporador superior:
$$\dot Q_{cond,B}=\dot Q_{evap,A}\ \Rightarrow\ \dot m_B(h_{2B}-h_{3B})=\dot m_A(h_{1A}-h_{4A}).$$
Cada ciclo: compresor isentrópico ($1\to2$), condensador ($2\to3$, líq. sat.), válvula ($3\to4$, $h_4=h_3$), evaporador ($4\to1$, vapor sat.).
\end{cardgear}
\vspace{1.2mm}
\begin{cardidea}{Paso 2 --- (a) Flujos másicos}{\faListOl}
Del efecto refrigerante del ciclo inferior y del acoplamiento:
$$\dot m_B=\frac{\dot Q_L}{h_{1B}-h_{4B}},\qquad \dot m_A=\frac{\dot m_B(h_{2B}-h_{3B})}{h_{1A}-h_{4A}}.$$
\end{cardidea}
\end{frame}

% ---- P9: Solución (potencia, COP) ----
\begin{frame}{P9 --- Solución paso a paso (potencia y COP)}
\begin{cardgear}{Paso 3 --- (b) y (c) Potencia y COP}{\faListOl}
La potencia total es la de ambos compresores; el COP se mide contra la carga fría:
$$\dot W=\dot m_B(h_{2B}-h_{1B})+\dot m_A(h_{2A}-h_{1A}),\qquad \mathrm{COP}_{cascada}=\frac{\dot Q_L}{\dot W}.$$
El COP de la cascada supera al de un solo ciclo entre $-30$ y $40\,^\circ$C, porque cada etapa opera con menor relación de presiones.
\end{cardgear}
\vspace{1.2mm}
\begin{cardnota}[\faExclamationTriangle]
{\footnotesize \textbf{Datos pendientes:} el valor numérico exige las \textbf{tablas de saturación y sobrecalentado del R-12} para leer $h$ en los $8$ estados ($-30$, $-15$, $-10$, $40\,^\circ$C y las salidas isentrópicas). El enunciado escaneado es ambiguo en las temperaturas/capacidades intermedias (aparecen ``$-10\,^\circ$C, $15$ TON'' y ``$-15\,^\circ$C''); conviene confirmarlo con el original. La \textbf{estructura y ecuaciones} de arriba resuelven el problema una vez fijados esos $h$.}
\end{cardnota}
\end{frame}
